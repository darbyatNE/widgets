# S&P 500 Regression Trend Lines

Each line is a least-squares regression fit through the monthly S&P 500 closes, from its **start month** through the **latest reading**. Start months run from the first month in the data through December 2016. Use the controls to choose which lines are drawn.

- **Linear:** a straight line in index points. Poor over long spans, because the index compounds.
- **Exponential (log-linear):** constant percentage growth, a straight line on a log axis.
- **Log-quadratic (default):** ln(price) = a + b·t + c·t². This lets the growth rate speed up or slow down over the window, so it tracks the "middle" of the price history more closely than a single constant rate. *Growth @ latest* is each fit's annual % growth rate at the latest reading. *Avg growth* is the fitted line's compound annual growth rate (CAGR) over its window.

```js
const raw = await FileAttachment("data/sp500_monthly.csv").text();
const data = d3.csvParseRows(raw)
  .filter((r) => r.length >= 2 && r[0])
  .map(([d, v]) => ({date: new Date(d + "T00:00:00"), close: +v.replace(/,/g, "")}))
  .filter((d) => !isNaN(d.date) && isFinite(d.close))
  .sort((a, b) => a.date - b.date);

const endPoint = data[data.length - 1];
const LAST_START = new Date("2016-12-31T23:59:59");
const fmtMonth = d3.utcFormat("%b %Y");
const fmtDate = d3.timeFormat("%m/%d/%Y");
const fmtNum = d3.format(",.2f");

// Candidate regression start months, independent of fit type so the picker keeps its selection.
const starts = data
  .map((d, i) => ({i, startDate: d.date, year: d.date.getFullYear(), key: d3.timeFormat("%Y-%m")(d.date)}))
  .filter((s) => s.startDate <= LAST_START);
const allKeys = starts.map((s) => s.key);
```

```js
const FITS = {
  "Linear": {degree: 1, log: false},
  "Exponential (log-linear)": {degree: 1, log: true},
  "Log-quadratic (changing growth rate)": {degree: 2, log: true}
};
const YEAR = 365.25 * 864e5;

// Least-squares polynomial fit of y (or ln y) against time. Time is centered and scaled so the normal equations stay well conditioned.
function fit(points, {degree, log}) {
  const t0 = points[0].date;
  const ts = points.map((p) => (p.date - t0) / YEAR);
  const ys = points.map((p) => (log ? Math.log(p.close) : p.close));
  const mid = (ts[0] + ts[ts.length - 1]) / 2;
  const half = Math.max((ts[ts.length - 1] - ts[0]) / 2, 1e-9);
  const xs = ts.map((t) => (t - mid) / half);

  const n = degree + 1;
  const A = Array.from({length: n}, () => new Array(n + 1).fill(0));
  for (let i = 0; i < xs.length; i++) {
    const pow = Array.from({length: 2 * n - 1}, (_, k) => xs[i] ** k);
    for (let r = 0; r < n; r++) {
      for (let c = 0; c < n; c++) A[r][c] += pow[r + c];
      A[r][n] += ys[i] * pow[r];
    }
  }
  for (let c = 0; c < n; c++) {
    let p = c;
    for (let r = c + 1; r < n; r++) if (Math.abs(A[r][c]) > Math.abs(A[p][c])) p = r;
    [A[c], A[p]] = [A[p], A[c]];
    for (let r = 0; r < n; r++) {
      if (r === c) continue;
      const f = A[r][c] / A[c][c];
      for (let k = c; k <= n; k++) A[r][k] -= f * A[c][k];
    }
  }
  const coef = A.map((row, k) => row[n] / row[k]);

  const at = (t) => coef.reduce((s, c, k) => s + c * ((t - mid) / half) ** k, 0);
  // d(at)/dt in units per year.
  const slopeAt = (t) => coef.reduce((s, c, k) => (k ? s + (k * c * ((t - mid) / half) ** (k - 1)) / half : s), 0);
  const predict = (date) => {
    const y = at((date - t0) / YEAR);
    return log ? Math.exp(y) : y;
  };

  const my = d3.mean(ys);
  let ssRes = 0, ssTot = 0, sqPct = 0;
  for (let i = 0; i < ts.length; i++) {
    const yhat = at(ts[i]);
    ssRes += (ys[i] - yhat) ** 2;
    ssTot += (ys[i] - my) ** 2;
    const price = points[i].close;
    sqPct += ((price - (log ? Math.exp(yhat) : yhat)) / price) ** 2;
  }
  const tEnd = (endPoint.date - t0) / YEAR;
  // Annual % growth of the fitted line at a date. For linear fits this is slope / fitted level,
  // i.e. the % gain over the next year; undefined where the line is at or below zero.
  const growthAt = (date) => {
    const t = (date - t0) / YEAR;
    if (log) return Math.exp(slopeAt(t)) - 1;
    const level = at(t);
    return level > 0 ? slopeAt(t) / level : NaN;
  };
  // Average annual % growth (CAGR) of the fitted line from the start month to the latest reading.
  const years = tEnd - ts[0];
  const yStart = at(ts[0]), yEnd = at(tEnd);
  const avgGrowth =
    years <= 0 ? NaN
    : log ? Math.exp((yEnd - yStart) / years) - 1
    : yStart > 0 && yEnd > 0 ? (yEnd / yStart) ** (1 / years) - 1
    : NaN;
  return {
    predict,
    growthAt,
    avgGrowth,
    r2: ssTot === 0 ? 1 : 1 - ssRes / ssTot,
    // RMS of percentage residuals in price space: comparable across all fit types.
    rmsPct: Math.sqrt(sqPct / ts.length),
    growthNow: growthAt(endPoint.date)
  };
}
```

```js
const model = FITS[fitType];
const useLog = model.log;
// Only the exponential fit has a constant % rate; linear and log-quadratic rates vary over time.
const isCurvedRate = !(model.log && model.degree === 1);
const regressions = starts.map(({i, key, year, startDate}) => {
    const pts = data.slice(i);
    const f = fit(pts, model);
    // Straight lines need only their endpoints; curved fits are sampled every month.
    const curve = (model.log || model.degree > 1 ? pts : [pts[0], pts[pts.length - 1]])
      .map((p) => ({key, date: p.date, value: f.predict(p.date)}));
    return {
      key,
      year,
      startDate,
      n: pts.length,
      slopePerYear: f.growthNow,
      avgGrowth: f.avgGrowth,
      growthAt: f.growthAt,
      r2: f.r2,
      rmsPct: f.rmsPct,
      fittedEnd: f.predict(endPoint.date),
      predict: f.predict,
      i,
      curve
    };
  });
```

```js
// Year-grouped checkbox picker. Exposes .value (array of "YYYY-MM" keys) for use with view().
function linePicker(regs, initial) {
  const selected = new Set(initial);
  const root = html`<div class="lp">
    <div class="lp-actions">
      <button type="button" data-act="all">Select all</button>
      <button type="button" data-act="none">Clear all</button>
      <button type="button" data-act="quarter">Quarterly starts</button>
      <button type="button" data-act="jan">January starts only</button>
      <span class="lp-count"></span>
    </div>
    <div class="lp-grid"></div>
  </div>`;
  const grid = root.querySelector(".lp-grid");
  const count = root.querySelector(".lp-count");
  const byYear = d3.group(regs, (r) => r.year);
  const monthBoxes = [];
  const yearBoxes = [];

  for (const [year, rows] of byYear) {
    const row = html`<div class="lp-row"><label class="lp-year"><input type="checkbox"> ${year}</label></div>`;
    const yearBox = row.querySelector("input");
    const boxes = rows.map((r) => {
      const lab = html`<label class="lp-month" title=${fmtMonth(r.startDate)}><input type="checkbox" value=${r.key}>${d3.timeFormat("%b")(r.startDate)}</label>`;
      const box = lab.querySelector("input");
      box.addEventListener("change", () => {
        box.checked ? selected.add(r.key) : selected.delete(r.key);
        emit();
      });
      monthBoxes.push(box);
      row.append(lab);
      return box;
    });
    yearBox.addEventListener("change", () => {
      for (const b of boxes) yearBox.checked ? selected.add(b.value) : selected.delete(b.value);
      emit();
    });
    yearBoxes.push({yearBox, boxes});
    grid.append(row);
  }

  function sync() {
    for (const b of monthBoxes) b.checked = selected.has(b.value);
    for (const {yearBox, boxes} of yearBoxes) {
      const on = boxes.filter((b) => b.checked).length;
      yearBox.checked = on === boxes.length;
      yearBox.indeterminate = on > 0 && on < boxes.length;
    }
    count.textContent = `${selected.size} of ${regs.length} lines selected`;
  }
  function emit() {
    sync();
    root.dispatchEvent(new Event("input", {bubbles: true}));
  }

  root.querySelector(".lp-actions").addEventListener("click", (e) => {
    const act = e.target.dataset?.act;
    if (!act) return;
    selected.clear();
    if (act === "all") regs.forEach((r) => selected.add(r.key));
    if (act === "jan") regs.filter((r) => r.startDate.getMonth() === 0).forEach((r) => selected.add(r.key));
    if (act === "quarter") regs.filter((r) => r.startDate.getMonth() % 3 === 0).forEach((r) => selected.add(r.key));
    emit();
  });

  Object.defineProperty(root, "value", {get: () => allKeys.filter((k) => selected.has(k))});
  sync();
  return root;
}
```

**Fit quality.** *RMS error* is the root-mean-square percentage gap between each monthly close and its fitted value. It is measured the same way for every fit type, so it's the fairest comparison. R² is computed on the fitted scale (price for Linear, ln(price) for the log fits), so it isn't comparable across fit types.

```js
// Median RMS error per model across the selected start months.
const comparison = Object.entries(FITS).map(([name, m]) => {

  return {"Fit type": name, "Median RMS error": errs.length ? d3.format(".1%")(d3.median(errs)) : "—", "Lines": errs.length};
});
```
